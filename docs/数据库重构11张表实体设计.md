1. Area.cs

区域采用自关联树结构：

using System.ComponentModel.DataAnnotations;

namespace YourProject.Domain.Entities;

这里我故意使用：

string? Value

而不是：

object Value

因为 EF Core 对 object 直接映射数据库并不理想。

例如：

Bool
Int
Float
String

统一存储成字符串，然后由 DataType 决定如何解释。

后面如果你追求高性能，我更建议设计成：

BoolValue
IntValue
DoubleValue
StringValue

或者针对时序数据库重新设计。

12. DbContext 配置

我不建议只依靠 EF Core 的约定。

你的系统关系比较复杂，最好显式配置。

例如：

using Microsoft.EntityFrameworkCore;
using YourProject.Domain.Entities;

namespace YourProject.Infrastructure.Data;

public class ScadaDbContext : DbContext
{
    public ScadaDbContext(
        DbContextOptions<ScadaDbContext> options)
        : base(options)
    {
    }

    public DbSet<Area> Areas => Set<Area>();

    public DbSet<Device> Devices => Set<Device>();

    public DbSet<Controller> Controllers => Set<Controller>();

    public DbSet<DeviceConnection> DeviceConnections => Set<DeviceConnection>();

    public DbSet<DataModel> DataModels => Set<DataModel>();

    public DbSet<DataPoint> DataPoints => Set<DataPoint>();

    public DbSet<DataPointMapping> DataPointMappings => Set<DataPointMapping>();

    public DbSet<DeviceDataModel> DeviceDataModels => Set<DeviceDataModel>();

    public DbSet<DeviceDataValue> DeviceDataValues => Set<DeviceDataValue>();

    public DbSet<DataHistory> DataHistories => Set<DataHistory>();

    public DbSet<AlarmRecord> AlarmRecords => Set<AlarmRecord>();

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        ConfigureArea(modelBuilder);
        ConfigureDevice(modelBuilder);
        ConfigureController(modelBuilder);
        ConfigureDeviceConnection(modelBuilder);
        ConfigureDataModel(modelBuilder);
        ConfigureDataPoint(modelBuilder);
        ConfigureDataPointMapping(modelBuilder);
        ConfigureDeviceDataModel(modelBuilder);
        ConfigureDeviceDataValue(modelBuilder);
        ConfigureDataHistory(modelBuilder);
        ConfigureAlarmRecord(modelBuilder);
    }

    private static void ConfigureArea(ModelBuilder builder)
    {
        builder.Entity<Area>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => x.Code)
                .IsUnique();

            entity.HasIndex(x => x.ParentId);

            entity.HasOne(x => x.Parent)
                .WithMany(x => x.Children)
                .HasForeignKey(x => x.ParentId)
                .OnDelete(DeleteBehavior.Restrict);

            entity.HasMany(x => x.Devices)
                .WithOne(x => x.Area)
                .HasForeignKey(x => x.AreaId)
                .OnDelete(DeleteBehavior.Restrict);
        });
    }

    private static void ConfigureDevice(ModelBuilder builder)
    {
        builder.Entity<Device>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => x.Code)
                .IsUnique();

            entity.HasIndex(x => x.AreaId);

            entity.HasIndex(x => x.ControllerId);

            entity.HasIndex(x => x.ParentId);

            entity.HasOne(x => x.Parent)
                .WithMany(x => x.Children)
                .HasForeignKey(x => x.ParentId)
                .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(x => x.Controller)
                .WithMany(x => x.Devices)
                .HasForeignKey(x => x.ControllerId)
                .OnDelete(DeleteBehavior.SetNull);
        });
    }

    private static void ConfigureController(ModelBuilder builder)
    {
        builder.Entity<Controller>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => x.Code)
                .IsUnique();
        });
    }

    private static void ConfigureDeviceConnection(ModelBuilder builder)
    {
        builder.Entity<DeviceConnection>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => x.ControllerId);

            entity.HasOne(x => x.Controller)
                .WithMany(x => x.Connections)
                .HasForeignKey(x => x.ControllerId)
                .OnDelete(DeleteBehavior.Cascade);
        });
    }

    private static void ConfigureDataModel(ModelBuilder builder)
    {
        builder.Entity<DataModel>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.Code,
                x.Version
            })
            .IsUnique();
        });
    }

    private static void ConfigureDataPoint(ModelBuilder builder)
    {
        builder.Entity<DataPoint>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.DataModelId,
                x.Code
            })
            .IsUnique();

            entity.HasOne(x => x.DataModel)
                .WithMany(x => x.DataPoints)
                .HasForeignKey(x => x.DataModelId)
                .OnDelete(DeleteBehavior.Cascade);
        });
    }

    private static void ConfigureDataPointMapping(ModelBuilder builder)
    {
        builder.Entity<DataPointMapping>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.DeviceId,
                x.DataPointId
            })
            .IsUnique();

            entity.HasIndex(x => x.ConnectionId);

            entity.HasOne(x => x.Device)
                .WithMany(x => x.DataPointMappings)
                .HasForeignKey(x => x.DeviceId)
                .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(x => x.DataPoint)
                .WithMany(x => x.Mappings)
                .HasForeignKey(x => x.DataPointId)
                .OnDelete(DeleteBehavior.Restrict);

            entity.HasOne(x => x.Connection)
                .WithMany(x => x.DataPointMappings)
                .HasForeignKey(x => x.ConnectionId)
                .OnDelete(DeleteBehavior.Restrict);
        });
    }

    private static void ConfigureDeviceDataModel(ModelBuilder builder)
    {
        builder.Entity<DeviceDataModel>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.DeviceId,
                x.DataModelId
            })
            .IsUnique();

            entity.HasOne(x => x.Device)
                .WithMany(x => x.DataModels)
                .HasForeignKey(x => x.DeviceId)
                .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(x => x.DataModel)
                .WithMany(x => x.Devices)
                .HasForeignKey(x => x.DataModelId)
                .OnDelete(DeleteBehavior.Restrict);
        });
    }

    private static void ConfigureDeviceDataValue(ModelBuilder builder)
    {
        builder.Entity<DeviceDataValue>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.DeviceId,
                x.DataPointId
            })
            .IsUnique();

            entity.HasOne(x => x.Device)
                .WithMany(x => x.DataValues)
                .HasForeignKey(x => x.DeviceId)
                .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(x => x.DataPoint)
                .WithMany(x => x.DeviceValues)
                .HasForeignKey(x => x.DataPointId)
                .OnDelete(DeleteBehavior.Restrict);
        });
    }

    private static void ConfigureDataHistory(ModelBuilder builder)
    {
        builder.Entity<DataHistory>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.DeviceId,
                x.DataPointId,
                x.Timestamp
            });

            entity.HasIndex(x => x.Timestamp);

            entity.HasOne(x => x.Device)
                .WithMany(x => x.DataHistories)
                .HasForeignKey(x => x.DeviceId)
                .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(x => x.DataPoint)
                .WithMany(x => x.Histories)
                .HasForeignKey(x => x.DataPointId)
                .OnDelete(DeleteBehavior.Restrict);
        });
    }

    private static void ConfigureAlarmRecord(ModelBuilder builder)
    {
        builder.Entity<AlarmRecord>(entity =>
        {
            entity.HasKey(x => x.Id);

            entity.HasIndex(x => new
            {
                x.DeviceId,
                x.StartTime
            });

            entity.HasIndex(x => x.Status);

            entity.HasOne(x => x.Device)
                .WithMany(x => x.AlarmRecords)
                .HasForeignKey(x => x.DeviceId)
                .OnDelete(DeleteBehavior.Cascade);

            entity.HasOne(x => x.DataPoint)
                .WithMany(x => x.AlarmRecords)
                .HasForeignKey(x => x.DataPointId)
                .OnDelete(DeleteBehavior.SetNull);
        });
    }
}

13. 最终数据库关系

现在这 11 张表的关系就非常清晰了：

Area
                          │
                    1     │     N
                          ↓
                       Device
                     /    │    \
                    /     │     \
                   ↓      ↓      ↓
             Controller  DataModel  Parent
                 │          │
                 │          │
                 ↓          ↓
        DeviceConnection  DataPoint
                 │          │
                 │          │
                 └─────┬────┘
                       ↓
               DataPointMapping
                       │
                       ↓
                  PLC / OPC UA


Device
   │
   ├──── DeviceDataModel ──── DataModel
   │
   ├──── DeviceDataValue ──── DataPoint
   │
   ├──── DataHistory ──────── DataPoint
   │
   └──── AlarmRecord ──────── DataPoint


---

一个地方我建议你现在就调整

虽然这 11 张表可以直接使用，但考虑到你之前规划的是 1000～10000 台 PLC/设备，我不太建议现在就把这套设计当成最终数据库结构。

尤其是下面三个地方值得在下一步认真处理：

① DataHistory.Value 不建议长期使用 string

如果 10000 台设备大量采集：

Float
Int
Bool

全部转换成字符串，历史表会比较浪费空间。

更合理的是：

DataHistory
├── DoubleValue
├── LongValue
├── BoolValue
└── StringValue

或者单独采用时序数据库。

② DataPointMapping 现在假设一个 DataPoint 只有一个映射

目前：

Device + DataPoint = 唯一

这个适合第一版。

但是以后可能出现：

主PLC
备用PLC
OPC UA

多个数据源对应同一个 DataPoint。

这时候需要增加：

Priority
IsPrimary

等字段。

③ Device.ControllerId 可能最终也需要拆掉

现在：

Device
   ↓
Controller

假设一个设备只有一个 PLC。

但工业现场可能出现：

包装机
├── Siemens PLC
├── Robot Controller
└── Safety PLC

一个设备实际上可能由多个 Controller 协同控制。

那时候最好变成：

Device
   │
   └── DeviceController
             │
             ├── PLC
             ├── Robot
             └── Safety PLC

所以这 11 张表可以作为第一版非常好的基础，但如果你的目标真的是做成 1000～10000 台设备的 SCADA 平台，我建议下一步直接把这套 EF Core 模型升级成“生产级数据库模型”：包括 EntityTypeConfiguration 分离、枚举、软删除、并发控制、审计字段、索引策略，以及把 DeviceController 加进去。
